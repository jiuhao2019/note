* 嵌套循环一个难觉察的错误
** 一个嵌套循环的例子
#+begin_src C
      while (/* 外层循环条件 */) {
          // 外层循环的其他操作
          {
              // 内层循环使用局部变量，每次进入内层循环时都会重新初始化
              unsigned int var = some_initial_value;
              while (var) {  // 当 var 不为 0 时执行内层循环
                  // 内层循环体代码

                  var--;  // 安全递减：当 var 为 0 时循环条件判断阻止进入循环体
              }
          }
          // 外层循环其他操作
      }
#+end_src
