uint8_t flash_is_write_protected(void)
{
    FLASH_OBProgramInitTypeDef ob;

    HAL_FLASH_Unlock();
    HAL_FLASH_OB_Unlock();

    HAL_FLASHEx_OBGetConfig(&ob);

    HAL_FLASH_OB_Lock();
    HAL_FLASH_Lock();
    if (ob.WRPPage != 0xffffffff)
    {
        return 1;   // 有区域被写保护
    }

    return 0;       // 无写保护
}
